-- invoice_setup.sql (Supabase → SQL Editor → New query → 贴上全部 → Run) -- 可以重复运行，不会删掉已有资料。 -- 1. 谁是管理员：把邮箱改成你的管理员账号 (app 里 RT_ADMIN_EMAIL 那个)。多个就用逗号分开。 create or replace function public.rt_is_admin() returns boolean language sql stable as $$ select lower(coalesce(auth.jwt() ->> 'email', '')) in ('testing@gmail.com') $$; -- 2. 客户资料表 create table if not exists public.invoice_clients ( id uuid primary key default gen_random_uuid(), name text not null, name_key text not null unique, bill_to_html text not null, customer_id uuid, created_at timestamptz not null default now(), updated_at timestamptz not null default now() ); -- 3. Invoice 表 create table if not exists public.invoices ( id uuid primary key default gen_random_uuid(), invoice_no text not null unique, client_name text, po_no text, invoice_date date not null, total numeric(12,2) not null default 0, snapshot jsonb, pdf_url text, created_at timestamptz not null default now(), updated_at timestamptz not null default now() ); -- (如果之前运行过旧版，这行会拿掉旧的 job_id 栏) alter table public.invoices drop column if exists job_id; -- 4. 只有管理员可以读写 alter table public.invoice_clients enable row level security; alter table public.invoices enable row level security; drop policy if exists "admin all invoice_clients" on public.invoice_clients; create policy "admin all invoice_clients" on public.invoice_clients for all using (public.rt_is_admin()) with check (public.rt_is_admin()); drop policy if exists "admin all invoices" on public.invoices; create policy "admin all invoices" on public.invoices for all using (public.rt_is_admin()) with check (public.rt_is_admin()); -- 5. 下一个编号：年(2位) + 月(2位) + 当月第几张(3位)，例如 2610001 -- 想要前缀 (INV2610001) 就把 v_prefix 改成 'INV' create or replace function public.next_invoice_no(p_date date) returns text language plpgsql security definer set search_path = public as $$ declare v_prefix text := ''; v_ym text := to_char(p_date, 'YYMM'); v_n int; begin if not public.rt_is_admin() then raise exception 'not allowed'; end if; select coalesce(max(substring(invoice_no from length(v_prefix || v_ym) + 1)::int), 0) + 1 into v_n from public.invoices where invoice_no ~ ('^' || v_prefix || v_ym || '[0-9]+$'); return v_prefix || v_ym || lpad(v_n::text, 3, '0'); end $$; -- 6. 保存 invoice：p_invoice_no 是空的 = 新的一张 (占用下一个编号)；有编号 = 更新那一张 drop function if exists public.save_invoice(text, text, text, date, numeric, jsonb); create or replace function public.save_invoice( p_invoice_no text, p_client_name text, p_po_no text, p_invoice_date date, p_total numeric, p_snapshot jsonb ) returns text language plpgsql security definer set search_path = public as $$ declare v_no text := nullif(p_invoice_no, ''); begin if not public.rt_is_admin() then raise exception 'not allowed'; end if; perform pg_advisory_xact_lock(hashtext('invoice_no')); if v_no is null then v_no := public.next_invoice_no(p_invoice_date); insert into public.invoices (invoice_no, client_name, po_no, invoice_date, total, snapshot) values (v_no, p_client_name, p_po_no, p_invoice_date, p_total, p_snapshot); else update public.invoices set client_name = p_client_name, po_no = p_po_no, invoice_date = p_invoice_date, total = p_total, snapshot = p_snapshot, updated_at = now() where invoice_no = v_no; if not found then raise exception 'invoice % not found', v_no; end if; end if; return v_no; end $$; notify pgrst, 'reload schema'; -- 7. 检查：下面应该显示两个表名 (不是空的) select to_regclass('public.invoice_clients') as clients_table, to_regclass('public.invoices') as invoices_table;
